          {/* Paso 2: Detalles del Servicio */}
          {bookingStep === 'details' && (
            <div className="space-y-6">
              {/* Resumen de la reserva */}
              <Card className="shadow-sm">
                <CardHeader className="pb-4">
                  <CardTitle className="flex items-center gap-2 text-lg">
                    <CheckCircle className="w-5 h-5 text-green-600" />
                    Reserva Confirmada
                  </CardTitle>
                  <CardDescription>
                    Tu fecha y hora han sido reservadas
                  </CardDescription>
                </CardHeader>
                <CardContent>
                  <div className="bg-green-50 border border-green-200 rounded-lg p-4">
                    <div className="grid grid-cols-1 md:grid-cols-3 gap-4 text-sm">
                      <div>
                        <p className="text-gray-600 mb-1">Fecha</p>
                        <p className="font-medium">
                          {selectedDate?.toLocaleDateString('es-ES', { 
                            weekday: 'long', 
                            day: 'numeric', 
                            month: 'long' 
                          })}
                        </p>
                      </div>
                      <div>
                        <p className="text-gray-600 mb-1">Hora</p>
                        <p className="font-medium">{selectedTimeSlot?.start}</p>
                      </div>
                      <div>
                        <p className="text-gray-600 mb-1">Duración</p>
                        <p className="font-medium">{formatDuration(selectedDuration)}</p>
                      </div>
                    </div>
                  </div>
                </CardContent>
              </Card>

              {/* Formulario de detalles del servicio */}
              <Card className="shadow-sm">
                <CardHeader className="pb-4">
                  <CardTitle className="flex items-center gap-2 text-lg">
                    <MessageSquare className="w-5 h-5 text-blue-600" />
                    Detalles del Servicio
                  </CardTitle>
                  <CardDescription>
                    Proporciona la información necesaria para el servicio
                  </CardDescription>
                </CardHeader>
                <CardContent className="space-y-6">
                  {/* Dirección del servicio */}
                  <div className="space-y-2">
                    <Label htmlFor="address" className="flex items-center gap-2">
                      <MapPin className="w-4 h-4" />
                      Dirección donde se realizará el servicio *
                    </Label>
                    <Input
                      id="address"
                      placeholder="Ej: Av. Providencia 1234, Piso 5, Depto 502"
                      value={serviceDetails.address}
                      onChange={(e) => setServiceDetails(prev => ({
                        ...prev,
                        address: e.target.value
                      }))}
                      required
                    />
                    <p className="text-xs text-gray-500">
                      {user?.address && serviceDetails.address === user.address 
                        ? "✓ Usando tu dirección registrada" 
                        : "Puedes usar otra dirección diferente a la de tu cuenta"}
                    </p>
                  </div>

                  {/* Teléfono de contacto */}
                  <div className="space-y-2">
                    <Label htmlFor="phone" className="flex items-center gap-2">
                      <Phone className="w-4 h-4" />
                      Teléfono de contacto *
                    </Label>
                    <Input
                      id="phone"
                      placeholder="Ej: +56 9 8888 7777"
                      value={serviceDetails.phone}
                      onChange={(e) => setServiceDetails(prev => ({
                        ...prev,
                        phone: e.target.value
                      }))}
                      required
                    />
                    <p className="text-xs text-gray-500">
                      {user?.phone && serviceDetails.phone === user.phone 
                        ? "✓ Usando tu teléfono registrado" 
                        : "El profesional te contactará a este número"}
                    </p>
                  </div>

                  {/* Descripción del problema/trabajo */}
                  <div className="space-y-2">
                    <Label htmlFor="description">
                      Describe el trabajo que necesitas *
                    </Label>
                    <textarea
                      id="description"
                      placeholder={`Describe detalladamente el trabajo de ${professional.service.toLowerCase()} que necesitas...`}
                      className="w-full p-3 border border-gray-300 rounded-lg resize-none h-24 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-transparent"
                      value={serviceDetails.description}
                      onChange={(e) => setServiceDetails(prev => ({
                        ...prev,
                        description: e.target.value
                      }))}
                      required
                    />
                    <p className="text-xs text-gray-500">
                      Incluye todos los detalles posibles para que el profesional pueda prepararse adecuadamente
                    </p>
                  </div>

                  {/* Solicitudes especiales */}
                  <div className="space-y-2">
                    <Label htmlFor="specialRequests">
                      Solicitudes especiales (Opcional)
                    </Label>
                    <textarea
                      id="specialRequests"
                      placeholder="Ej: Traer materiales específicos, trabajar en horario silencioso, usar entrada trasera..."
                      className="w-full p-3 border border-gray-300 rounded-lg resize-none h-20 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-transparent"
                      value={serviceDetails.specialRequests}
                      onChange={(e) => setServiceDetails(prev => ({
                        ...prev,
                        specialRequests: e.target.value
                      }))}
                    />
                  </div>

                  {/* Botones de navegación */}
                  <div className="flex flex-col sm:flex-row gap-3 pt-6 border-t">
                    <Button 
                      variant="outline" 
                      onClick={() => setBookingStep('datetime')}
                      className="flex-1"
                    >
                      <ArrowLeft className="w-4 h-4 mr-2" />
                      Volver a Fecha y Hora
                    </Button>
                    <Button 
                      onClick={() => setBookingStep('payment')}
                      disabled={!serviceDetails.address || !serviceDetails.phone || !serviceDetails.description}
                      className="flex-1"
                    >
                      Continuar al Pago
                      <ArrowLeft className="w-4 h-4 ml-2 rotate-180" />
                    </Button>
                  </div>
                </CardContent>
              </Card>
            </div>
          )}

          {/* Paso 3: Confirmación y Pago */}
          {bookingStep === 'payment' && (
            <div className="space-y-6">
              {/* Resumen completo de la reserva */}
              <Card className="shadow-sm">
                <CardHeader className="pb-4">
                  <CardTitle className="flex items-center gap-2 text-lg">
                    <CreditCard className="w-5 h-5 text-blue-600" />
                    Confirmar y Pagar
                  </CardTitle>
                  <CardDescription>
                    Revisa todos los detalles antes de proceder al pago
                  </CardDescription>
                </CardHeader>
                <CardContent className="space-y-6">
                  {/* Información del servicio */}
                  <div className="bg-gray-50 rounded-lg p-4">
                    <h4 className="font-semibold mb-3">Resumen del Servicio</h4>
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4 text-sm">
                      <div className="space-y-2">
                        <div>
                          <p className="text-gray-600">Servicio</p>
                          <p className="font-medium">{professional.service}</p>
                        </div>
                        <div>
                          <p className="text-gray-600">Profesional</p>
                          <p className="font-medium">{professional.name}</p>
                        </div>
                        <div>
                          <p className="text-gray-600">Fecha</p>
                          <p className="font-medium">
                            {selectedDate?.toLocaleDateString('es-ES', { 
                              weekday: 'long', 
                              day: 'numeric', 
                              month: 'long' 
                            })}
                          </p>
                        </div>
                      </div>
                      <div className="space-y-2">
                        <div>
                          <p className="text-gray-600">Hora</p>
                          <p className="font-medium">{selectedTimeSlot?.start}</p>
                        </div>
                        <div>
                          <p className="text-gray-600">Duración</p>
                          <p className="font-medium">{formatDuration(selectedDuration)}</p>
                        </div>
                        <div>
                          <p className="text-gray-600">Total</p>
                          <p className="font-bold text-green-600 text-xl">${currentPrice.toLocaleString()}</p>
                        </div>
                      </div>
                    </div>
                  </div>

                  {/* Detalles del servicio */}
                  <div className="space-y-3">
                    <h4 className="font-semibold">Detalles del Servicio</h4>
                    <div className="bg-blue-50 rounded-lg p-4 space-y-3 text-sm">
                      <div>
                        <p className="text-gray-600 mb-1">Dirección</p>
                        <p className="font-medium">{serviceDetails.address}</p>
                      </div>
                      <div>
                        <p className="text-gray-600 mb-1">Teléfono de contacto</p>
                        <p className="font-medium">{serviceDetails.phone}</p>
                      </div>
                      <div>
                        <p className="text-gray-600 mb-1">Descripción del trabajo</p>
                        <p className="font-medium">{serviceDetails.description}</p>
                      </div>
                      {serviceDetails.specialRequests && (
                        <div>
                          <p className="text-gray-600 mb-1">Solicitudes especiales</p>
                          <p className="font-medium">{serviceDetails.specialRequests}</p>
                        </div>
                      )}
                    </div>
                  </div>

                  {/* Términos y condiciones */}
                  <div className="bg-yellow-50 border border-yellow-200 rounded-lg p-4">
                    <div className="flex items-start gap-3">
                      <AlertCircle className="w-5 h-5 text-yellow-600 mt-0.5" />
                      <div className="text-sm">
                        <p className="font-medium text-yellow-800 mb-1">Términos de la Reserva</p>
                        <ul className="text-yellow-700 space-y-1 text-xs">
                          <li>• El pago se procesará de forma segura a través de MercadoPago</li>
                          <li>• El profesional te contactará dentro de las próximas 2 horas</li>
                          <li>• Puedes cancelar hasta 4 horas antes del servicio</li>
                          <li>• Los materiales adicionales se cobran aparte según el trabajo</li>
                        </ul>
                      </div>
                    </div>
                  </div>

                  {/* Botones de navegación */}
                  <div className="flex flex-col sm:flex-row gap-3 pt-6 border-t">
                    <Button 
                      variant="outline" 
                      onClick={() => setBookingStep('details')}
                      className="flex-1"
                    >
                      <ArrowLeft className="w-4 h-4 mr-2" />
                      Volver a Detalles
                    </Button>
                    <Button 
                      onClick={handleBooking}
                      disabled={isBooking}
                      className="flex-1 bg-green-600 hover:bg-green-700"
                    >
                      {isBooking ? (
                        <>
                          <div className="animate-spin rounded-full h-4 w-4 border-b-2 border-white mr-2"></div>
                          Procesando...
                        </>
                      ) : (
                        <>
                          <CreditCard className="w-4 h-4 mr-2" />
                          Pagar ${currentPrice.toLocaleString()}
                        </>
                      )}
                    </Button>
                  </div>
                </CardContent>
              </Card>
            </div>
          )}
